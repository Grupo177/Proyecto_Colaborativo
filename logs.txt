from excepciones import ServicioNoDisponibleError, DatosInvalidosError
from logger import log_error, log_info

def procesar_solicitud(datos):
    try:
        log_info("Iniciando procesamiento de solicitud...")
        
        # Simulación de validación
        if not datos:
            raise DatosInvalidosError("entrada", "La lista de datos está vacía.")
            
        # Simulación de fallo de servicio
        raise ServicioNoDisponibleError("El servidor de reservas no responde.")

    except DatosInvalidosError as de:
        print(f"Advertencia: {de.mensaje}")
        log_error(de)
    except ServicioNoDisponibleError as se:
        print(f"Error de conexión: {se.mensaje}")
        log_error(se)
    except Exception as e:
        print(f"Error inesperado: {e}")
        log_error(f"Inesperado: {e}")
    else:
        print("Procesamiento completado con éxito.")
    finally:
        # Esto garantiza que el sistema siempre esté listo para la siguiente tarea
        print("Sistema listo para la siguiente operación.")
